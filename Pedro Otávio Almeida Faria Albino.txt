Exercicio02

public class Exercicio02 {

    public static void main(String[] args) {
        Scanner ler = new Scanner(System.in);

        int vitorias = 0;

        System.out.println("Digite os resultados das 6 partidas.");
        System.out.println("Use V para vitória e P para derrota.");

        for (int i = 1; i <= 6; i++) {

            System.out.println("Digite o resultado da partida " + i + ":");
            char resultado = ler.next().charAt(0);

            if (resultado == 'V') {
                vitorias++;
            }
        }

        System.out.println("Total de vitórias: " + vitorias);

        if (vitorias >= 5) {
            System.out.println("Grupo: 1");
        }
        else if (vitorias >= 3) {
            System.out.println("Grupo: 2");
        }
        else if (vitorias >= 1) {
            System.out.println("Grupo: 1");
        }
        else {
            System.out.println("Grupo: -1");
        }
    }
}

Exercicio03

public class Exercicio03 {

    public static void main(String[] args) {
        Scanner ler = new Scanner(System.in);

        System.out.println("Digite o valor de S:");
        int s = ler.nextInt();

        System.out.println("Digite o início do intervalo A:");
        int a = ler.nextInt();

        System.out.println("Digite o final do intervalo B:");
        int b = ler.nextInt();

        int quantidade = 0;

        for (int i = a; i <= b; i++) {

            int numero = i;
            int soma = 0;

            while (numero > 0) {

                soma = soma + (numero % 10);

                numero = numero / 10;
            }

            if (soma == s) {
                quantidade++;
            }
        }

        System.out.println("Quantidade de números encontrados: " + quantidade);
    }
}
